```python
def quick_sort(q, l, r):
    if l >= r:
        return
    i, j = l, r
    pivot = q[(l + r) // 2]
    while i <= j:
        while q[i] < pivot:
            i += 1
        while q[j] > pivot:
            j -= 1
        if i <= j: # 一定是≦
            q[i], q[j] = q[j], q[i]
            i += 1
            j -= 1
    quick_sort(q, l, j)
    quick_sort(q, i, r)

```
